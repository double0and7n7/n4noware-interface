local function New(className, properties, parent)
	local object = Instance.new(className)

	for property, value in pairs(properties or {}) do
		object[property] = value
	end

	object.Parent = parent

	return object
end

local function Corner(object, radius)
	local corner = Instance.new("UICorner")
	corner.CornerRadius = UDim.new(0, radius)
	corner.Parent = object

	return corner
end

local function Stroke(object, transparency, thickness)
	local stroke = Instance.new("UIStroke")

	stroke.Color = Theme.Border
	stroke.Transparency = transparency or 0
	stroke.Thickness = thickness or 1

	stroke.Parent = object

	return stroke
end

local function Tween(object, duration, properties, style, direction)
	return TweenService:Create(
		object,
		TweenInfo.new(
			duration,
			style or Enum.EasingStyle.Quart,
			direction or Enum.EasingDirection.Out
		),
		properties
	)
end

local function IsPointer(input)
	return input.UserInputType == Enum.UserInputType.MouseButton1
		or input.UserInputType == Enum.UserInputType.Touch
end

local function CreateNavigationButton(text)
	local button = New("TextButton", {
		Size = UDim2.fromOffset(104, 38),
		BackgroundColor3 = Theme.Surface,
		BorderSizePixel = 0,
		Text = text,
		TextColor3 = Theme.Muted,
		TextSize = 10,
		Font = Enum.Font.GothamMedium,
		AutoButtonColor = false,
		ZIndex = 6
	}, Navigation)

	Corner(button, 8)
	Stroke(button, 0.5)

	local indicator = New("Frame", {
		Name = "Indicator",
		Size = UDim2.fromOffset(3, 16),
		Position = UDim2.new(0, 5, 0.5, -8),
		BackgroundColor3 = Theme.Accent,
		BackgroundTransparency = 1,
		BorderSizePixel = 0,
		ZIndex = 7
	}, button)

	Corner(indicator, 3)

	button.MouseEnter:Connect(function()
		if ActiveButton ~= button then
			Tween(button, 0.13, {
				BackgroundColor3 = Theme.SurfaceHover
			}):Play()
		end
	end)

	button.MouseLeave:Connect(function()
		if ActiveButton ~= button then
			Tween(button, 0.13, {
				BackgroundColor3 = Theme.Surface
			}):Play()
		end
	end)

	button.Activated:Connect(function()
		if ActiveButton and ActiveButton ~= button then
			local oldIndicator = ActiveButton:FindFirstChild("Indicator")

			Tween(ActiveButton, 0.15, {
				BackgroundColor3 = Theme.Surface,
				TextColor3 = Theme.Muted
			}):Play()

			if oldIndicator then
				Tween(oldIndicator, 0.15, {
					BackgroundTransparency = 1
				}):Play()
			end
		end

		ActiveButton = button

		Tween(button, 0.18, {
			BackgroundColor3 = Theme.AccentDark,
			TextColor3 = Theme.Text
		}):Play()

		Tween(indicator, 0.18, {
			BackgroundTransparency = 0
		}):Play()
	end)

	return button
end

function Components.Toggle(parent, title, default, callback)
	local value = default == true

	local button = New("TextButton", {
		Size = UDim2.new(1, 0, 0, 40),
		BackgroundColor3 = Theme.Surface,
		BorderSizePixel = 0,
		Text = "",
		AutoButtonColor = false,
		ZIndex = 30
	}, parent)

	Corner(button, 8)
	Stroke(button, 0.4)

	Components.Label(button, title, {
		Size = UDim2.new(1, -65, 1, 0),
		Position = UDim2.fromOffset(11, 0),
		ZIndex = 31
	})

	local track = New("Frame", {
		Size = UDim2.fromOffset(40, 21),
		Position = UDim2.new(1, -51, 0.5, -10),
		BackgroundColor3 = Theme.Border,
		BorderSizePixel = 0,
		ZIndex = 31
	}, button)

	Corner(track, 12)

	local knob = New("Frame", {
		Size = UDim2.fromOffset(15, 15),
		Position = UDim2.fromOffset(3, 3),
		BackgroundColor3 = Theme.Muted,
		BorderSizePixel = 0,
		ZIndex = 32
	}, track)

	Corner(knob, 9)

	local function Render()
		Tween(track, 0.2, {
			BackgroundColor3 = value and Theme.AccentDark or Theme.Border
		}):Play()

		Tween(knob, 0.23, {
			Position = value
				and UDim2.fromOffset(22, 3)
				or UDim2.fromOffset(3, 3),

			BackgroundColor3 = value
				and Theme.Accent
				or Theme.Muted
		}):Play()
	end

	button.Activated:Connect(function()
		value = not value
		Render()

		if callback then
			callback(value)
		end
	end)

	Render()

	return {
		Object = button,

		Get = function()
			return value
		end,

		Set = function(newValue)
			value = newValue == true
			Render()
		end
	}
end

function Components.Dropdown(parent, title, options, defaultIndex, callback)
	local selected = math.clamp(defaultIndex or 1, 1, #options)
	local open = false

	local holder = New("Frame", {
		Size = UDim2.new(1, 0, 0, 40),
		BackgroundTransparency = 1,
		ZIndex = 30
	}, parent)

	local button = New("TextButton", {

		Text = "",
		AutoButtonColor = false,
		ZIndex = 31
	}, holder)

	Corner(button, 8)
	Stroke(button, 0.4)

	Components.Label(button, title, {
		Size = UDim2.new(0.48, 0, 1, 0),
		Position = UDim2.fromOffset(11, 0),
		ZIndex = 32
	})

	local selectedText = Components.Label(button, options[selected], {
		Size = UDim2.new(0.42, 0, 1, 0),
		Position = UDim2.new(0.48, 0, 0, 0),
		Alignment = Enum.TextXAlignment.Right,
		TextColor = Theme.Accent,
		ZIndex = 32
	})

	local arrow = Components.Label(button, "⌄", {
		Size = UDim2.fromOffset(20, 40),
		Position = UDim2.new(1, -25, 0, 0),
		Alignment = Enum.TextXAlignment.Center,
		TextColor = Theme.Muted,
		ZIndex = 32
	})

	local list = New("Frame", {
		Size = UDim2.new(1, 0, 0, 0),
		Position = UDim2.fromOffset(0, 45),
		BackgroundColor3 = Theme.Window,
		BorderSizePixel = 0,
		Visible = false,
		ClipsDescendants = true,
		ZIndex = 60
	}, holder)

	Corner(list, 8)
	Stroke(list, 0.2)

	local layout = New("UIListLayout", {
		Padding = UDim.new(0, 3),
		SortOrder = Enum.SortOrder.LayoutOrder
	}, list)

	local function Select(index)
		selected = index
		selectedText.Text = options[index]

		if callback then
			callback(options[index], index)
		end
	end

	for index, option in ipairs(options) do
		local optionButton = New("TextButton", {
			Size = UDim2.new(1, -8, 0, 32),
			Position = UDim2.fromOffset(4, 0),
			BackgroundColor3 = Theme.Surface,
			BorderSizePixel = 0,
			Text = option,
			TextColor3 = Theme.Text,
			TextSize = 10,
			Font = Enum.Font.GothamMedium,
			AutoButtonColor = false,
			LayoutOrder = index,
			ZIndex = 61
		}, list)

		Corner(optionButton, 6)

		optionButton.MouseEnter:Connect(function()
			Tween(optionButton, 0.12, {
				BackgroundColor3 = Theme.SurfaceHover
			}):Play()
		end)

		optionButton.MouseLeave:Connect(function()
			Tween(optionButton, 0.12, {
				BackgroundColor3 = Theme.Surface
			}):Play()
		end)

		optionButton.Activated:Connect(function()
			Select(index)

			open = false
			arrow.Text = "⌄"

			Tween(list, 0.18, {
				Size = UDim2.new(1, 0, 0, 0)
			}):Play()

			task.delay(0.18, function()
				if not open then
					list.Visible = false
				end
			end)
		end)
	end

	button.Activated:Connect(function()
		open = not open

		if open then
			list.Visible = true
			list.Size = UDim2.new(1, 0, 0, 0)

			local height = math.min(#options * 35 + 5, 145)

			Tween(list, 0.24, {
				Size = UDim2.new(1, 0, 0, height)
			}):Play()

			arrow.Text = "⌃"
		else
			Tween(list, 0.18, {
				Size = UDim2.new(1, 0, 0, 0)
			}):Play()

			task.delay(0.18, function()
				if not open then
					list.Visible = false
				end
			end)

			arrow.Text = "⌄"
		end
	end)

	return {
		Object = holder,

		Get = function()
			return options[selected]
		end,

		Set = function(index)
			if options[index] then
				Select(index)
			end
		end
	}
end

--==================================================
-- DIRECTION DETECTION
--==================================================

function PopoutManager.CalculatePosition(data)
	local camera = workspace.CurrentCamera

	if not camera then
		return Vector2.new(10, 10), "right"
	end

	local viewport = camera.ViewportSize

	local windowPosition = Window.AbsolutePosition
	local windowSize = Window.AbsoluteSize

	local width = data.Width
	local height = data.Height

	local gap = 10
	local margin = 10

	local rightX = windowPosition.X + windowSize.X + gap
	local leftX = windowPosition.X - width - gap

	local bottomY = windowPosition.Y + windowSize.Y - 50
	local topY = windowPosition.Y - height + 50

	local rightSpace = viewport.X - rightX
	local leftSpace = windowPosition.X

	local bottomSpace = viewport.Y - bottomY
	local topSpace = windowPosition.Y

	if rightSpace >= width + margin then
		return Vector2.new(
			rightX,
			math.clamp(
				windowPosition.Y + 50,
				margin,
				viewport.Y - height - margin
			)
		), "right"
	end

	if leftSpace >= width + margin then
		return Vector2.new(
			leftX,
			math.clamp(
				windowPosition.Y + 50,
				margin,
				viewport.Y - height - margin
			)
		), "left"
	end

	local centeredX = math.clamp(
		windowPosition.X + windowSize.X / 2 - width / 2,
		margin,
		viewport.X - width - margin
	)

	if bottomSpace >= height + margin then
		return Vector2.new(centeredX, bottomY), "bottom"
	end

	if topSpace >= height + margin then
		return Vector2.new(centeredX, topY), "top"
	end

	if rightSpace >= leftSpace then
		return Vector2.new(
			math.clamp(rightX, margin, viewport.X - width - margin),
			math.clamp(windowPosition.Y + 50, margin, viewport.Y - height - margin)
		), "right"
	end

	return Vector2.new(
		math.clamp(leftX, margin, viewport.X - width - margin),
		math.clamp(windowPosition.Y + 50, margin, viewport.Y - height - margin)
	), "left"
end

function PopoutManager.Position(data, instant)
	if data.Detached then
		return
	end

	local position, direction =
		PopoutManager.CalculatePosition(data)

	data.Side = direction

	if instant then
		data.Frame.Position = UDim2.fromOffset(
			position.X,
			position.Y
		)

		return
	end

	Tween(data.Frame, 0.25, {
		Position = UDim2.fromOffset(
			position.X,
			position.Y
		)
	}, Enum.EasingStyle.Quint):Play()
end

--==================================================
-- POPOUT DRAGGING
--==================================================

function PopoutManager.EnableDragging(data)
	local frame = data.Frame
	local header = data.Header

	local dragging = false
	local dragStart
	local startPosition

	header.InputBegan:Connect(function(input)
		if not IsPointer(input) then
			return
		end

		dragging = true
		data.Detached = true

		dragStart = input.Position
		startPosition = frame.Position

		frame.ZIndex = 100
	end)

	UIS.InputChanged:Connect(function(input)
		if not dragging then
			return
		end

		if input.UserInputType ~= Enum.UserInputType.MouseMovement
			and input.UserInputType ~= Enum.UserInputType.Touch then
			return
		end

		local delta = input.Position - dragStart

		frame.Position = UDim2.new(
			startPosition.X.Scale,
			startPosition.X.Offset + delta.X,
			startPosition.Y.Scale,
			startPosition.Y.Offset + delta.Y
		)
	end)

	UIS.InputEnded:Connect(function(input)
		if IsPointer(input) then
			dragging = false
		end
	end)
end

--==================================================
-- SHOW / HIDE
--==================================================

function PopoutManager.Hide(data)
	if not data then
		return
	end

	local frame = data.Frame

	Tween(frame, 0.2, {
		Size = UDim2.fromOffset(0, data.Height)
	}, Enum.EasingStyle.Quint):Play()

	task.delay(0.2, function()
		frame.Visible = false
	end)

	if PopoutManager.Current == data then
		PopoutManager.Current = nil
	end
end

function PopoutManager.Show(data)
	if PopoutManager.Current == data then
		PopoutManager.Hide(data)
		return
	end

	if PopoutManager.Current then
		PopoutManager.Hide(PopoutManager.Current)
	end

	PopoutManager.Current = data

	data.Detached = false

	local frame = data.Frame

	frame.Visible = true
	frame.Size = UDim2.fromOffset(0, data.Height)

	PopoutManager.Position(data, true)

	-- slightly springy opening
	Tween(frame, 0.34, {
		Size = UDim2.fromOffset(data.Width + 7, data.Height)
	}, Enum.EasingStyle.Back):Play()

	task.delay(0.3, function()
		Tween(frame, 0.12, {
			Size = UDim2.fromOffset(data.Width, data.Height)
		}):Play()
	end)
end

--==================================================
-- GLOBAL INPUT
--==================================================

UIS.InputChanged:Connect(function(input)
	if input.UserInputType ~= Enum.UserInputType.MouseMovement
		and input.UserInputType ~= Enum.UserInputType.Touch then
		return
	end

	if WindowDragging then
		local delta = input.Position - WindowDragStart

		Window.Position = UDim2.new(
			WindowStartPosition.X.Scale,
			WindowStartPosition.X.Offset + delta.X,
			WindowStartPosition.Y.Scale,
			WindowStartPosition.Y.Offset + delta.Y
		)

		if PopoutManager.Current
			and not PopoutManager.Current.Detached then

			PopoutManager.Position(
				PopoutManager.Current,
				false
			)
		end
	end

	if Resizing then
		local delta = input.Position - ResizeStart

		local width = math.max(
			MinimumWidth,
			OriginalSize.X + delta.X
		)

		local height = math.max(
			MinimumHeight,
			OriginalSize.Y + delta.Y
		)

		Window.Size = UDim2.fromOffset(
			width,
			height
		)

		if PopoutManager.Current
			and not PopoutManager.Current.Detached then

			PopoutManager.Position(
				PopoutManager.Current,
				false
			)
		end
	end
end)

UIS.InputEnded:Connect(function(input)
	if IsPointer(input) then
		WindowDragging = false
		Resizing = false
	end
end)
